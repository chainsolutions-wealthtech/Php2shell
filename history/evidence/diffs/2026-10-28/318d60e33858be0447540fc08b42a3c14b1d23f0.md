+"""Governed, offline review of candidate PHP advisory catalog updates.
+
+No network calls, no file writes and no source verification claims. Candidate
+files must pass the same strict snapshot schema validation as the baseline.
+Results describe changes for human review; they do NOT approve, fetch, merge,
+publish or change the active advisory catalog.
+"""
+from datetime import date
+from hashlib import sha256
+import json
+import re
+
+from .security_advisories import load_advisory_catalog
+
+_SENSITIVE_FIELDS = frozenset({
+    "fixed_upstream", "ghsa", "severity", "cvss_v4_base_score",
+    "published_on", "sources",
+})
+
+
+def _as_of_date(as_of_utc):
+    if not isinstance(as_of_utc, str) or not re.fullmatch(r"\d{4}-\d{2}-\d{2}", as_of_utc):
+        raise ValueError("as_of_utc must be an ISO calendar date YYYY-MM-DD")
+    try:
+        return date.fromisoformat(as_of_utc)
+    except ValueError as exc:
+        raise ValueError("Invalid as_of_utc calendar date") from exc
+
+
+def _fingerprint(catalog):
+    canonical = json.dumps(catalog, ensure_ascii=False, sort_keys=True,
+                           separators=(",", ":")).encode("utf-8")
+    return sha256(canonical).hexdigest()
+
+
+def _advisory_changes(before, after):
+    changed = sorted(key for key in (set(before) | set(after))
+                     if before.get(key) != after.get(key))
+    return changed
+
+
+def _threshold_changes(before, after):
+    # Report only version/branch deltas, not a diagnosis or proof of risk.
+    old = before.get("fixed_upstream", {})
+    new = after.get("fixed_upstream", {})
+    return [
+        {
+            "branch": branch,
+            "previous_upstream_fix": old.get(branch),
+            "candidate_upstream_fix": new.get(branch),
+            "review_required": True,
+        }
+        for branch in sorted(set(old) | set(new))
+        if old.get(branch) != new.get(branch)
+    ]
+
+
+def propose_catalog_update(candidate_path, as_of_utc, baseline_path=None):
+    """Compare validated local snapshots with append-only safeguards.
+
+    A deletion, source revision, scope reduction, version change, or reverted
+    review date requires explicit independent upstream verification before
+    any change to the canonical snapshot. Nothing is applied by this function.
+    """
+    if not isinstance(candidate_path, (str, bytes)) and not hasattr(candidate_path, "__fspath__"):
+        raise ValueError("Candidate path is required")
+    observed = _as_of_date(as_of_utc)
+    baseline = load_advisory_catalog(baseline_path)
+    candidate = load_advisory_catalog(candidate_path)
+    baseline_date = date.fromisoformat(baseline["reviewed_on_utc"])
+    candidate_date = date.fromisoformat(candidate["reviewed_on_utc"])
+    if baseline_date > observed or candidate_date > observed:
+        raise ValueError("Catalog review date cannot be in the future")
+
+    old = {entry["cve"]: entry for entry in baseline["advisories"]}
+    new = {entry["cve"]: entry for entry in candidate["advisories"]}
+    added = sorted(set(new) - set(old))
+    removed = sorted(set(old) - set(new))
+    modified = []
+    for cve in sorted(set(old) & set(new)):
+        changed_fields = _advisory_changes(old[cve], new[cve])
+        if changed_fields:
+            modified.append({
+                "cve": cve,
+                "changed_fields": changed_fields,
+                "security_sensitive_change": bool(_SENSITIVE_FIELDS.intersection(changed_fields)),
+                "upstream_fix_changes": _threshold_changes(old[cve], new[cve]),
+            })
+
+    metadata_changes = sorted(
+        key for key in (set(baseline) | set(candidate)) - {"advisories"}
+        if baseline.get(key) != candidate.get(key)
+    )
+    previous_fingerprint = _fingerprint(baseline)
+    candidate_fingerprint = _fingerprint(candidate)
+    content_changed = previous_fingerprint != candidate_fingerprint
+    if not content_changed:
+        proposal_state = "NO_CHANGE"
+    elif removed or candidate_date < baseline_date:
+        proposal_state = "BLOCKED_DELETION_OR_DATE_ROLLBACK"
+    else:
+        proposal_state = "AWAITING_INDEPENDENT_UPSTREAM_REVIEW"
+
+    warnings = []
+    if removed:
+        warnings.append("Advisory removals are not permitted without a documented decision.")
+    if candidate_date < baseline_date:
+        warnings.append("Candidate review date is older than the current snapshot.")
+    if modified or added or metadata_changes:
+        warnings.append("Changed advisory data requires independent primary-source reverification.")
+    return {
+        "schema_version": "0.2",
+        "proposal_type": "OFFLINE_ADVISORY_UPDATE_REVIEW_NOT_APPLICATION",
+        "as_of_utc": as_of_utc,
+        "baseline_reviewed_on_utc": baseline["reviewed_on_utc"],
+        "candidate_reviewed_on_utc": candidate["reviewed_on_utc"],
+        "baseline_canonical_json_sha256": previous_fingerprint,
+        "candidate_canonical_json_sha256": candidate_fingerprint,
+        "content_changed": content_changed,
+        "proposal_state": proposal_state,
+        "added_cves": added,
+        "removed_cves": removed,
+        "modified_advisories": modified,
+        "changed_catalog_metadata_fields": metadata_changes,
+        "warnings": warnings,
+        "canonical_catalog_modified": False,
+        "independent_upstream_reverification_required": True,
+        "asset_applicability": "insufficient_evidence",
+        "limitations": (
+            "Read-only proposal using two local schema-valid JSON files. "
+            "URLs in those files are not contacted or cryptographically "
+            "authenticated. Upstream vendor fixes, downstream backports "
+            "and actual installed PHP are not attested; no patch is applied."
+        ),
+    }

+# V0.2 — Proposition gouvernée de mise à jour du catalogue CVE (hors ligne)
+
+**Statut : implémentation locale de comparaison et de revue, pas de collecte automatisée des avis upstream ni de publication effective.**
+
+## Objectif
+
+Permettre à un agent de comparer le **catalogue CVE actuellement acquis** à un candidat local (JSON) avant toute décision. Prévenir les suppressions silencieuses, les retours en arrière de date, les modifications de sources et les déplacements de seuils correctifs. Conserver l'audit existant `audit-catalog` et toutes les commandes précédentes.
+
+## Exemple
+
+```bash
+python -m app.cli advisories
+python -m app.cli audit-catalog --as-of 2026-10-28
+python -m app.cli propose-catalog-update --candidate ./candidate.json --as-of 2026-10-28
+```
+
+- Le candidat doit être un fichier JSON **local régulier, sans symlink**, inférieur au plafond du catalogue. Il suit les mêmes validations strictes que `app/data/php_security_advisories.json`.
+- La fonction `propose_catalog_update()` **n'écrit rien** dans le catalogue installé, ne contacte aucun domaine et n'ajoute pas de permission d'analyse. Elle renvoie les identifiants CVE ajoutés/supprimés, les champs modifiés, les versions upstream de référence avant/après, l'évolution de la date de revue et les empreintes SHA-256 des deux instantanés.
+- `NO_CHANGE` = contenu canonique identique ; `AWAITING_INDEPENDENT_UPSTREAM_REVIEW` = proposition modifiée non appliquée ; `BLOCKED_DELETION_OR_DATE_ROLLBACK` = retrait d'avis ou régression de date, à traiter séparément sous décision documentée.
+- Changer la source primaire PHP, le GHSA, le score, la sévérité ou un correctif upstream impose **une nouvelle validation externe réelle**. Le contrôle de format d'une URL officielle **ne démontre pas que cette URL existe ni qu'elle authentifie son contenu**.
+- Les empreintes SHA-256 permettent la comparaison de contenu, mais **ne constituent pas une signature éditeur**. L'automatisation ne doit pas transformer ce résultat en approbation automatique.
+
+## Tests de non-régression
+
+`tests/test_catalog_update.py` couvre : catalogue identique, ajout d'une CVE de fixture, suppression interdite, retour en arrière de date, changement de branche PHP, changement de lien CERT-FR, rejet des sources falsifiées, JSON invalide, symlink, date future et fonctionnement local sans réseau. Aucune fixture synthétique de ce test ne doit être traitée comme un avis réel.
+
+`python -m unittest discover -s tests -v` est exécuté dans GitHub Actions au HEAD exact.
+
+## Suite du programme
+
+Cette tranche prolonge **REQ-005**, sans la clore. Il reste notamment :
+- récupération *bornée et autorisée* des avis PHP/GHSA/CERT-FR à partir de sources vérifiables ;
+- contrôle de fraîcheur sur les sources distantes, changements officiels, signatures si disponibles et approbation de la mise à jour ;
+- interprétation des correctifs rétroportés par distribution/fournisseur et attestation runtime PHP/SAPI sous mandat P2 ;
+- diagnostic de l'actif et remise de ses propositions de correction à PHPPATCH, sans application automatique.
+
+**Interdiction :** aucun catalogue candidat n'est fusionné automatiquement, aucune CVE déclarée applicable au seul motif d'une version PHP fournie et aucun système tiers n'est analysé ou modifié par cette commande.
