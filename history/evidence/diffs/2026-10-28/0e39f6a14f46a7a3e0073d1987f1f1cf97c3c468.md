+"""Fail-closed URL intake preview for a future *authorized* PHP2SHELL crawler.
+
+Pure parsing: no DNS lookup, network call, SAPI detection, site scan, claim,
+mandate verification or permission grant. DNS/IP/redirect verification must
+be performed separately and freshly at EACH network connection.
+"""
+import ipaddress
+import re
+from urllib.parse import urlsplit
+
+from .core import Scope, Target
+
+_MANDATE = re.compile(r"[A-Za-z0-9][A-Za-z0-9._:-]{0,79}")
+_PATH = re.compile(r"/[A-Za-z0-9._~/-]*")
+_BLOCKED_SUFFIXES = (
+    ".localhost", ".local", ".internal", ".onion", ".invalid", ".test",
+)
+_BLOCKED_HOSTS = {"localhost", "metadata.google.internal"}
+_MAX_URL = 2048
+
+
+def preview_url_scope(url, scope="public_metadata", mandate_reference=None,
+                      max_pages=20, requests_per_minute=6):
+    """Validate a user-entered site URL without authorizing or fetching it.
+
+    No scanner should ever use this parse-only response as an authorization
+    or SSRF defense. DNS rebinding, CNAMEs, IPv4/IPv6 and redirects require
+    independent approved enforcement at connection time.
+    """
+    if (not isinstance(url, str) or not 1 <= len(url) <= _MAX_URL
+            or any(ord(char) < 33 or ord(char) == 127 for char in url)
+            or "\\" in url or "?" in url or "#" in url):
+        raise ValueError("Expected a clean URL without credentials, query or fragment")
+    try:
+        parts = urlsplit(url)
+        hostname = parts.hostname
+        port = parts.port
+    except ValueError as exc:
+        raise ValueError("Malformed URL") from exc
+    if (parts.scheme != "https" or not parts.netloc or not isinstance(hostname, str)
+            or parts.username is not None or parts.password is not None
+            or port not in (None, 443)):
+        raise ValueError("Only direct public HTTPS URLs on port 443 are accepted")
+    if (not hostname.isascii() or hostname.endswith(".")
+            or hostname in _BLOCKED_HOSTS
+            or hostname.endswith(_BLOCKED_SUFFIXES)):
+        raise ValueError("Local or reserved hostname is not a public scan target")
+    try:
+        ipaddress.ip_address(hostname)
+    except ValueError:
+        pass
+    else:
+        raise ValueError("IP literal cannot be entered as a crawler hostname")
+    if (all(label.isdigit() or label.lower().startswith("0x")
+            for label in hostname.split("."))):
+        raise ValueError("Ambiguous numeric IP-like hostname")
+    host = Target(hostname, Scope.PUBLIC).hostname
+    if "." not in host:
+        raise ValueError("Public site hostname requires a DNS suffix")
+    path = parts.path or "/"
+    if (not _PATH.fullmatch(path) or "//" in path
+            or ".." in path.split("/")):
+        raise ValueError("Unsafe or noncanonical URL path")
+    if (type(max_pages) is not int or not 1 <= max_pages <= 100
+            or type(requests_per_minute) is not int
+            or not 1 <= requests_per_minute <= 30):
+        raise ValueError("Invalid requested crawl budget")
+    try:
+        declared_scope = Scope(scope)
+    except (ValueError, TypeError) as exc:
+        raise ValueError("Unknown declared authorization scope") from exc
+    if mandate_reference is not None and (
+            not isinstance(mandate_reference, str)
+            or not _MANDATE.fullmatch(mandate_reference)):
+        raise ValueError("Invalid mandate reference")
+    if declared_scope == Scope.OWNED and not mandate_reference:
+        raise ValueError("Source review declaration requires a mandate reference")
+
+    # Do not return free text from the mandate input in the publicly rendered
+    # preview. Storage and permission proofs belong in a protected control plane.
+    return {
+        "schema_version": "0.2",
+        "preview_type": "OFFLINE_URL_SCOPE_PREVIEW_NOT_ADMISSION",
+        "normalized_url": "https://" + host + path,
+        "hostname": host,
+        "base_path": path,
+        "declared_scope": declared_scope.value,
+        "operator_mandate_supplied": bool(mandate_reference),
+        "mandate_authentication_verified": False,
+        "target_ownership_verified": False,
+        "dns_and_public_ip_validated": False,
+        "allowed_to_crawl": False,
+        "allowed_to_inspect_source": False,
+        "max_pages_requested": max_pages,
+        "requests_per_minute_requested": requests_per_minute,
+        "request_count_executed": 0,
+        "network_requests_made": False,
+        "next_gates": [
+            "Authenticate account, mandate, consent, scope and expiry.",
+            "Verify DNS CNAME/A/AAAA responses and every selected peer IP is public.",
+            "Resolve and pin each connection destination; reject DNS rebinding.",
+            "Enforce redirects within approved origin and revalidate every hop.",
+            "Apply robots policy, rate/concurrency budgets and cancellation.",
+            "Only then schedule an authorized passive crawl; never assume PHP source access.",
+            "Request an additional governed P2 source mandate before source inspection.",
+        ],
+        "limitations": (
+            "Parse-only URL preview. No DNS, HTTP, authorization, owner "
+            "attestation, PHP version or vulnerability check. A syntactically "
+            "valid HTTPS URL is not necessarily public or safe to visit."
+        ),
+    }
