+"""Short, honest anonymous summary of supplied public HTTP metadata.
+
+A *pure* formatter/validator: this does not send a request, authenticate a
+user, establish collector provenance, determine CVE exploitability, or save
+any report. A production public scanner must separately implement approved
+admission, DNS/SSRF guards, redirects, quotas and trusted evidence collection.
+"""
+from .url_intake import preview_url_scope
+
+
+_ALLOWED_HEADERS = frozenset({
+    "content-type",
+    "strict-transport-security",
+    "content-security-policy",
+    "x-content-type-options",
+    "referrer-policy",
+    "permissions-policy",
+})
+_REQUIRED = frozenset({
+    "schema_version", "evidence_type", "http_status",
+    "response_headers_present",
+})
+_RECOMMENDED = (
+    ("strict-transport-security",
+     "La présence de l'en-tête HSTS n'est pas indiquée dans les éléments fournis."),
+    ("content-security-policy",
+     "La présence d'une politique CSP n'est pas indiquée dans les éléments fournis."),
+    ("x-content-type-options",
+     "La présence de X-Content-Type-Options n'est pas indiquée dans les éléments fournis."),
+)
+
+
+def _validate_evidence(evidence):
+    if not isinstance(evidence, dict) or set(evidence) != _REQUIRED:
+        raise ValueError("Invalid unverified public evidence schema")
+    if (evidence["schema_version"] != "0.2"
+            or evidence["evidence_type"]
+            != "UNVERIFIED_SUPPLIED_PUBLIC_RESPONSE"):
+        raise ValueError("Unsupported public evidence")
+    status = evidence["http_status"]
+    if type(status) is not int or not 100 <= status <= 599:
+        raise ValueError("Invalid declared HTTP status")
+    headers = evidence["response_headers_present"]
+    if (not isinstance(headers, list) or len(headers) > len(_ALLOWED_HEADERS)
+            or len(headers) != len(set(str(x) for x in headers))
+            or any(not isinstance(h, str) or h not in _ALLOWED_HEADERS
+                   for h in headers)):
+        raise ValueError("Invalid public header-name indicators")
+    return status, set(headers)
+
+
+def build_guest_public_summary(url, public_response_evidence=None):
+    """Return ≤3 short indicators, without remote inspection or storage.
+
+    A supplied snapshot is NOT proof that it came from the URL, even if
+    the caller says it did. This function deliberately cannot return a
+    site security rating, detailed report, or authenticated ownership.
+    """
+    preview = preview_url_scope(url, scope="public_metadata")
+    result = {
+        "schema_version": "0.2",
+        "summary_type": "ANONYMOUS_SHORT_PREVIEW",
+        "hostname": preview["hostname"],
+        "status": "NOT_ANALYZED",
+        "headline": "Adresse acceptée. Aucune analyse réelle du site n'a été effectuée.",
+        "observations": [],
+        "site_contacted": False,
+        "network_requests_made": False,
+        "audit_performed": False,
+        "vulnerability_confirmed": False,
+        "site_ownership_verified": False,
+        "detailed_report_available": False,
+        "can_read_detailed_report": False,
+        "report_stored": False,
+        "account_required_for_report_storage": True,
+        "can_read_private_source": False,
+        "can_patch": False,
+        "can_deploy": False,
+        "next_action": "SIGN_IN_FOR_PERSISTENT_DETAILED_REPORT",
+        "limitations": (
+            "Aucun site contacté. Une URL seule ne constitue pas un audit. "
+            "Une analyse publique réelle nécessite une collecte admise et "
+            "contrôlée ; un rapport privé exige une connexion et des droits."
+        ),
+    }
+    if public_response_evidence is None:
+        return result
+
+    status, present = _validate_evidence(public_response_evidence)
+    observations = [{
+        "label": "Statut HTTP déclaré",
+        "message": "Un échantillon non vérifié indique le statut HTTP "
+                   + str(status) + ".",
+        "evidence_verified": False,
+    }]
+    for name, missing_notice in _RECOMMENDED:
+        if len(observations) >= 3:
+            break
+        if name not in present:
+            observations.append({
+                "label": "Information de sécurité à confirmer",
+                "message": missing_notice,
+                "evidence_verified": False,
+            })
+    result["status"] = "UNVERIFIED_SUPPLIED_EVIDENCE"
+    result["headline"] = (
+        "Aperçu limité fondé sur des informations fournies, non vérifiées "
+        "sur le site. Il ne s'agit pas d'un audit."
+    )
+    result["observations"] = observations
+    return result

+# P2S-UX-002 — Moteur initial de synthèse courte pour visiteur
+
+**État réel :** premier composant opérationnel **purement local**, sans collecteur HTTP, sans endpoint anonyme, sans stockage de rapport ni compte. La fonction `app.public_summary.build_guest_public_summary(url, public_response_evidence=None)` calcule une réponse courte, mais ne prétend jamais avoir audité le site.
+
+## Contrat
+
+- Entrée : URL HTTPS validée par `app.url_intake.preview_url_scope`. Un appel sans preuve retourne `NOT_ANALYZED`, aucun constat et une explication intelligible.
+- En option : un **échantillon de métadonnées HTTP fourni, non authentifié**, avec seulement un code de statut HTTP déclaré et la présence déclarée de certains noms d'en-têtes autorisés ; aucune valeur d'en-tête, donnée privée ni contenu arbitraire. Les clés et limites sont contrôlées strictement.
+- Sortie : au plus trois points clairs, **tous marqués comme preuves non vérifiées** ; aucun score de sécurité, aucune conclusion CVE, aucun statut de vulnérabilité prouvée. Les avertissements portent sur ce qui n'est pas indiqué dans l'échantillon fourni, pas sur l'état certain du site.
+- Aucune requête DNS/HTTP, aucune copie persistante, aucun rapport détaillé, aucun compte utilisateur, aucun accès au code source ou correction. Tous les champs d'exécution restent `false`.
+- L'action `SIGN_IN_FOR_PERSISTENT_DETAILED_REPORT` est **un jalon produit futur** : aucun écran de connexion, rapport détaillé ni sauvegarde n'est construit par ce composant.
+
+## Suite nécessaire pour le vrai parcours public
+
+1. Construire un **collecteur P1 public réellement autorisé** et protégé en profondeur (contrôles d'accès et d'abus, DNS SSRF au moment de chaque connexion, robots et restrictions, budgets, redirections, taille des réponses, annulation, journaux minimisés). Ne pas confondre le résumé d'une URL avec un audit.
+2. Brancher ce collecteur au moteur de synthèse, en conservant une preuve de provenance que le client ne peut pas forger ; seule une collecte réellement attestée peut être présentée comme réellement effectuée.
+3. Implémenter une authentification grand public réelle, des droits de visibilité, un stockage privé durable, et un rapport public détaillé uniquement sur les données effectivement obtenues.
+4. Vérifier séparément la propriété du domaine puis, facultativement, connecter GitHub avec permissions minimales pour analyse P2. PHPPATCH et déploiement restent gouvernés distinctement.
+
+**Aucune autorisation sur un site tiers ne se déduit du statut d'invité ou de l'inscription.** Ne pas publier un bouton d'audit actif tant que le collecteur et les contrôles sont absents.
