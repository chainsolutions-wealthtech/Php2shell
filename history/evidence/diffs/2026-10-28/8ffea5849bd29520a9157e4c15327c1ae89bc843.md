+"""HTML rendering regression tests: safe escaping, integrity and no network."""
+import json
+import socket
+import subprocess
+import sys
+import tempfile
+import unittest
+from pathlib import Path
+from unittest.mock import patch
+
+from app.audit_dossier import build_audit_dossier
+from app.report_html import render_dossier_html
+
+
+class HtmlReportTests(unittest.TestCase):
+    def setUp(self):
+        self.temp = tempfile.TemporaryDirectory()
+        self.addCleanup(self.temp.cleanup)
+        self.root = Path(self.temp.name)
+        (self.root / "index.php").write_text(
+            "<?php\n eval($untrusted);\n", encoding="utf-8")
+        (self.root / "php.ini").write_text(
+            "display_errors=On\npassword=CONFIG_SECRET_NEVER_SHOW\n")
+        self.dossier = build_audit_dossier(self.root, "2026-10-28")
+
+    def test_printable_html_has_no_scripts_or_server_paths(self):
+        with patch.object(socket.socket, "connect", side_effect=AssertionError("network")):
+            rendered = render_dossier_html(self.dossier)
+        self.assertIn("<!doctype html>", rendered)
+        self.assertIn('lang="fr"', rendered)
+        self.assertIn("Content-Security-Policy", rendered)
+        self.assertIn("default-src", rendered)
+        self.assertIn("PHP-DYNAMIC-EVAL", rendered)
+        self.assertIn("PHP-CONFIG-DISPLAY-ERRORS", rendered)
+        self.assertIn("Aucun patch autorisé", rendered)
+        self.assertNotIn("<script", rendered.lower())
+        self.assertNotIn(str(self.root), rendered)
+        self.assertNotIn("CONFIG_SECRET_NEVER_SHOW", rendered)
+        self.assertNotIn("$untrusted", rendered)
+        self.assertNotIn("https://", rendered)
+
+    def test_html_escapes_untrusted_source_filenames(self):
+        unusual = self.root / "x_<svg onload=alert(1)>.php"
+        unusual.write_text("<?php\n system($cmd);\n")
+        dossier = build_audit_dossier(self.root, "2026-10-28")
+        html = render_dossier_html(dossier)
+        self.assertNotIn("<svg", html)
+        self.assertIn("&lt;svg", html)
+        self.assertNotIn("<script", html)
+        self.assertNotIn("$cmd", html)
+
+    def test_refuses_forged_auth_or_hash_modified_dossiers(self):
+        forged = dict(self.dossier)
+        forged["production_safe_to_patch"] = True
+        with self.assertRaises(ValueError):
+            render_dossier_html(forged)
+        forged = dict(self.dossier)
+        forged["report_fingerprint_sha256"] = "0" * 64
+        with self.assertRaises(ValueError):
+            render_dossier_html(forged)
+        forged = dict(self.dossier)
+        forged["observations"] = list(self.dossier["observations"])
+        forged["observations"][0] = dict(forged["observations"][0],
+                                         asset_vulnerability_confirmed=True)
+        with self.assertRaises(ValueError):
+            render_dossier_html(forged)
+
+    def test_empty_report_is_never_certified(self):
+        with tempfile.TemporaryDirectory() as tmp:
+            dossier = build_audit_dossier(tmp, "2026-10-28")
+        output = render_dossier_html(dossier)
+        self.assertIn("ne constitue pas un certificat", output)
+        self.assertIn("Aucun patch autorisé", output)
+
+    def test_cli_html_format_without_json_regression(self):
+        base = [sys.executable, "-m", "app.cli", "audit-dossier",
+                "--path", str(self.root), "--as-of", "2026-10-28"]
+        response = subprocess.run(base + ["--format", "html"],
+                                  capture_output=True, text=True, check=True)
+        self.assertIn("<!doctype html>", response.stdout)
+        self.assertNotIn(str(self.root), response.stdout)
+        json_output = subprocess.run(base, capture_output=True, text=True, check=True)
+        self.assertEqual(json.loads(json_output.stdout)["schema_version"], "0.2")
