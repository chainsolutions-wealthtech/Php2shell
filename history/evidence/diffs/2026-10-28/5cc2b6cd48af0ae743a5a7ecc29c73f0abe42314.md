+"""Standalone printable HTML rendering of a strictly local audit dossier.
+
+No scripts, external resources or network operations. This is a document
+rendering helper, not the deployed PHP2SHELL or PHPPATCH frontend.
+"""
+from html import escape
+
+from .dossier_handoff import _verify_dossier
+
+_CSS = """
+:root{font-family:system-ui,-apple-system,BlinkMacSystemFont,"Segoe UI",sans-serif;
+color:#14243b;background:#f5f7fa}
+body{max-width:920px;margin:2rem auto;padding:0 1rem}
+header{background:#142d50;color:white;border-radius:12px;padding:1.5rem}
+main{margin-top:1.5rem}
+section{background:white;border:1px solid #dce3ec;border-radius:10px;
+padding:1.25rem;margin:1rem 0}
+h1{font-size:1.65rem;margin:0 0 .6rem}h2{font-size:1.2rem}
+p,li{line-height:1.5}small{color:#526174}
+.item{padding:1rem 0;border-top:1px solid #e0e6ee}
+code{overflow-wrap:anywhere}
+.status{font-weight:bold}
+@media print{body{max-width:none;margin:0;padding:0;background:white}
+header{background:#142d50!important;color:white!important;print-color-adjust:exact}
+section{break-inside:avoid;box-shadow:none}}
+"""
+
+
+def _val(value):
+    return escape(str(value), quote=True)
+
+
+def render_dossier_html(dossier):
+    """Refuse elevated or forged dossiers before rendering any user data."""
+    observations = _verify_dossier(dossier)
+    runtime = dossier.get("declared_runtime")
+    summary = dossier["summary"]
+    lines = [
+        "<!doctype html>",
+        '<html lang="fr"><head><meta charset="utf-8">',
+        '<meta name="viewport" content="width=device-width, initial-scale=1">',
+        '<meta http-equiv="Content-Security-Policy" content="default-src \'none\'; '
+        "style-src 'unsafe-inline'; base-uri 'none'; form-action 'none'\">",
+        "<title>PHP2SHELL — Audit local (brouillon)</title>",
+        "<style>" + _CSS + "</style></head><body>",
+        "<header>",
+        "<h1>PHP2SHELL — Dossier d’audit local</h1>",
+        "<p class='status'>Brouillon · Aucun diagnostic d’actif confirmé · "
+        "Aucun patch autorisé</p>",
+        "<p>Date de référence : " + _val(dossier["as_of_utc"]) + "</p>",
+        "</header><main>",
+        "<section aria-labelledby='resume'><h2 id='resume'>Résumé de couverture</h2>",
+        "<p>Fichiers PHP examinés : " + _val(summary["php_files_scanned"]) + "</p>",
+        "<p>Indices nécessitant revue : " + _val(len(observations)) + "</p>",
+        "<p>Cette analyse lexicale et locale ne démontre pas l’absence de vulnérabilité.</p>",
+        "</section>",
+        "<section aria-labelledby='observations'><h2 id='observations'>Constats à examiner</h2>",
+    ]
+    if not observations:
+        lines.append("<p>Aucun indice relevé. Cela ne constitue pas un certificat de sécurité.</p>")
+    for row in observations:
+        lines += [
+            '<article class="item">',
+            "<h3>" + _val(row["rule"]) + "</h3>",
+            "<p>Référence : <code>" + _val(row["evidence_id"]) + "</code></p>",
+            "<p>Source relative : <code>" +
+            _val(row["source_relative_path"]) + "</code> · Ligne " +
+            _val(row["source_line"]) + "</p>",
+            "<p>Priorité indicative : " + _val(row["severity_hint"]) + "</p>",
+            "<p>Proposition pour revue : " + _val(row["remediation_proposal"]) + "</p>",
+            "<small>Confiance : à examiner · Patch : non généré</small>",
+            "</article>",
+        ]
+    lines.append("</section>")
+    if runtime is not None:
+        lines += [
+            "<section aria-labelledby='runtime'><h2 id='runtime'>PHP/SAPI déclarés</h2>",
+            "<p>Hôte : " + _val(runtime["asset_hostname"]) + "</p>",
+            "<p>Version PHP : " + _val(runtime["php_version_declared"]) + "</p>",
+            "<p>SAPI : " + _val(runtime["sapi_declared"]) + "</p>",
+            "<p><strong>Identité, mandat, installation, signature et backports non attestés.</strong></p>",
+            "<ul>",
+        ]
+        for entry in runtime["comparisons"]:
+            lines.append(
+                "<li>" + _val(entry["cve"]) + " : " + _val(entry["status"]) +
+                " · seuil upstream " + _val(entry["reference_fix"]) +
+                " · applicabilité insuffisamment établie</li>"
+            )
+        lines += ["</ul>", "</section>"]
+    lines += [
+        "<section aria-labelledby='gates'><h2 id='gates'>Conditions préalables à PHPPATCH</h2>",
+        "<ol>",
+        *("<li>" + _val(step) + "</li>" for step in dossier["next_gates"]),
+        "</ol></section>",
+        "<section><h2>Limites</h2><p>" + _val(dossier["limitations"]) + "</p></section>",
+        "</main></body></html>",
+    ]
+    return "\n".join(lines)
