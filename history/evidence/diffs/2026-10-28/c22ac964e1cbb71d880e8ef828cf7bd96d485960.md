+"""Verify an EXTERNALLY signed P1/P2 attestation in a local offline contract.
+
+This module contains NO issuer, signer, trusted keys, MCP credentials, session,
+network client, authorization grant, scanner, patch or deployment code.
+
+A production control plane must supply pinned key material, trusted
+issuer/audience, request binding and a live revocation provider from its own
+secure configuration, NOT from an HTTP payload or the untrusted site owner.
+Even successful cryptographic verification is NOT admission to execute.
+"""
+import base64
+import binascii
+import json
+import re
+
+from cryptography.exceptions import InvalidSignature
+from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PublicKey
+
+
+_ATT_TYPE = "p2s-mcp-attestation+jwt"
+_HEADER = frozenset({"typ", "alg", "kid"})
+_CLAIMS = frozenset({
+    "iss", "aud", "sub", "project_id", "asset_hostname", "scope",
+    "action", "mandate_id", "iat", "nbf", "exp", "jti",
+})
+_BINDING = frozenset({
+    "subject", "project_id", "asset_hostname", "scope", "action",
+})
+_ID = re.compile(r"[A-Za-z0-9][A-Za-z0-9._:-]{0,79}\Z", re.ASCII)
+_B64 = re.compile(r"[A-Za-z0-9_-]+\Z", re.ASCII)
+_ACTIONS = {
+    "public_crawl": "public_metadata",
+    "source_read": "owned_source",
+    "patch_propose": "owned_source",
+    "patch_apply": "owned_source",
+    "deploy": "owned_source",
+}
+_MAX_TOKEN_CHARS = 4096
+_MAX_ATTESTATION_SECONDS = 300
+
+
+def _deny():
+    # Errors must never echo JWTs, signed claims or secure configuration.
+    raise ValueError("Untrusted or invalid external attestation")
+
+
+def _decode_b64url(fragment, *, max_decoded=3000):
+    if (not isinstance(fragment, str) or len(fragment) > 4000
+            or not _B64.fullmatch(fragment)):
+        _deny()
+    try:
+        value = base64.b64decode(
+            fragment + "=" * (-len(fragment) % 4), altchars=b"-_",
+            validate=True,
+        )
+    except (ValueError, binascii.Error):
+        _deny()
+    if len(value) > max_decoded or base64.urlsafe_b64encode(value).rstrip(
+            b"=").decode("ascii") != fragment:
+        _deny()
+    return value
+
+
+def _strict_json(fragment):
+    def unique(pairs):
+        obj = {}
+        for key, value in pairs:
+            if key in obj:
+                _deny()
+            obj[key] = value
+        return obj
+
+    try:
+        obj = json.loads(_decode_b64url(fragment).decode("utf-8"),
+                         object_pairs_hook=unique)
+    except (ValueError, UnicodeError, TypeError):
+        _deny()
+    if not isinstance(obj, dict):
+        _deny()
+    return obj
+
+
+def _valid_id(value):
+    return isinstance(value, str) and bool(_ID.fullmatch(value))
+
+
+def verify_external_attestation(token, *, trusted_public_keys,
+                                expected_issuer, expected_audience,
+                                request_binding, observed_at_epoch,
+                                revocation_checker):
+    """Validate a signed external claim; ALWAYS prohibit execution.
+
+    trusted_public_keys and the revocation checker are supplied by the
+    *trusted backend*; never expose them as caller-supplied request fields.
+    This code does not discover trust by accepting untrusted JWT 'jku',
+    'jwk', 'x5u' or algorithms, and does not manufacture MCP rights.
+    """
+    if (not isinstance(token, str)
+            or len(token) > _MAX_TOKEN_CHARS
+            or token.count(".") != 2
+            or type(observed_at_epoch) is not int
+            or observed_at_epoch < 0
+            or not _valid_id(expected_issuer)
+            or not _valid_id(expected_audience)
+            or not isinstance(trusted_public_keys, dict)
+            or not callable(revocation_checker)
+            or not isinstance(request_binding, dict)
+            or set(request_binding) != _BINDING):
+        _deny()
+
+    parts = token.split(".")
+    header, claims = _strict_json(parts[0]), _strict_json(parts[1])
+    if (set(header) != _HEADER or header.get("typ") != _ATT_TYPE
+            or header.get("alg") != "EdDSA"
+            or not _valid_id(header.get("kid"))
+            or set(claims) != _CLAIMS):
+        _deny()
+
+    kid = header["kid"]
+    raw = trusted_public_keys.get(kid)
+    if not isinstance(raw, bytes) or len(raw) != 32:
+        _deny()
+    signature = _decode_b64url(parts[2], max_decoded=128)
+    if len(signature) != 64:
+        _deny()
+    try:
+        Ed25519PublicKey.from_public_bytes(raw).verify(
+            signature, (parts[0] + "." + parts[1]).encode("ascii")
+        )
+    except (InvalidSignature, ValueError, TypeError):
+        _deny()
+
+    for field in ("iss", "aud", "sub", "project_id", "asset_hostname",
+                  "scope", "action", "mandate_id", "jti"):
+        if not _valid_id(claims[field]):
+            _deny()
+    if (claims["iss"] != expected_issuer
+            or claims["aud"] != expected_audience
+            or claims["scope"] not in ("public_metadata", "owned_source")
+            or claims["action"] not in _ACTIONS
+            or _ACTIONS[claims["action"]] != claims["scope"]):
+        _deny()
+    for key, claim_field in (
+        ("subject", "sub"), ("project_id", "project_id"),
+        ("asset_hostname", "asset_hostname"), ("scope", "scope"),
+        ("action", "action")
+    ):
+        if (not _valid_id(request_binding[key])
+                or request_binding[key] != claims[claim_field]):
+            _deny()
+
+    iat, nbf, exp = (claims["iat"], claims["nbf"], claims["exp"])
+    if (any(type(x) is not int for x in (iat, nbf, exp))
+            or not (0 <= iat <= nbf <= observed_at_epoch < exp)
+            or exp - iat > _MAX_ATTESTATION_SECONDS):
+        _deny()
+
+    try:
+        revoked = revocation_checker(claims["jti"])
+    except Exception:
+        _deny()
+    if type(revoked) is not bool or revoked:
+        _deny()
+
+    return {
+        "schema_version": "0.2",
+        "verification_type": "EXTERNAL_ED25519_ATTESTATION_NOT_ADMISSION",
+        "decision": "VERIFIED_ATTESTATION_NOT_EXECUTION_ADMISSION",
+        "signature_verified": True,
+        "claims_bound_to_request": True,
+        "revocation_checked": True,
+        "authorized": False,
+        "execution_enabled": False,
+        "mcp_claim_verified": False,
+        "user_session_authenticated": False,
+        "project_acl_verified": False,
+        "ready_for_production": False,
+        "limitations": (
+            "Cryptographic claim verification only. The verifier cannot "
+            "grant P1/P2 authority, prove a user session, verify MCP runtime "
+            "claims, authorize a scanner or patch, or deploy anything. "
+            "External issuer onboarding, key rotation and live revocation "
+            "must be governed before production integration."
+        ),
+    }

+
+The test signer is ephemeral and test-only; no keys, JWTs, OAuth sessions,
+privileges or production service are minted by the application.
+"""

+import json
+import unittest
+from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey
+from cryptography.hazmat.primitives import serialization
+
+from app.signed_attestation import verify_external_attestation
+
+
+def b64(raw):
+    return base64.urlsafe_b64encode(raw).rstrip(b"=").decode("ascii")
+
+
+class SignedAttestationContractTests(unittest.TestCase):
+    def setUp(self):
+        self.signer = Ed25519PrivateKey.generate()
+        self.public_key = self.signer.public_key().public_bytes(
+            serialization.Encoding.Raw, serialization.PublicFormat.Raw
+        )
+        self.keys = {"mcp-key-01": self.public_key}
+        self.expected = {
+            "subject": "operator-1", "project_id": "P2S-PROJECT-001",
+            "asset_hostname": "site.example.com", "scope": "owned_source",
+            "action": "source_read",
+        }
+        self.claims = {
+            "iss": "mcp-trusted-authority",
+            "aud": "php2shell-local-attestation-contract",
+            "sub": "operator-1",
+            "project_id": "P2S-PROJECT-001",
+            "asset_hostname": "site.example.com",
+            "scope": "owned_source",
+            "action": "source_read",
+            "mandate_id": "P2S-MANDATE-001",
+            "iat": 1793163600, "nbf": 1793163600,
+            "exp": 1793163720, "jti": "P2S-ATTEST-001",
+        }
+        self.now = 1793163650
+
+    def token(self, claims=None, header=None, *, tamper_signature=False):
+        head = {"typ": "p2s-mcp-attestation+jwt", "alg": "EdDSA",
+                "kid": "mcp-key-01"} if header is None else header
+        body = self.claims if claims is None else claims
+        signed = (b64(json.dumps(head, separators=(",", ":")).encode())
+                  + "." + b64(json.dumps(body, separators=(",", ":")).encode()))
+        signature = self.signer.sign(signed.encode("ascii"))
+        if tamper_signature:
+            signature = signature[:-1] + bytes([signature[-1] ^ 1])
+        return signed + "." + b64(signature)
+
+    def verify(self, token=None, expected=None, now=None, *,
+               keys=None, revoked=None, issuer=None, audience=None):
+        return verify_external_attestation(
+            token if token is not None else self.token(),
+            trusted_public_keys=self.keys if keys is None else keys,
+            expected_issuer=issuer if issuer is not None else self.claims["iss"],
+            expected_audience=(audience if audience is not None
+                               else self.claims["aud"]),
+            request_binding=self.expected if expected is None else expected,
+            observed_at_epoch=self.now if now is None else now,
+            revocation_checker=(revoked if revoked is not None
+                                else lambda jti: False),
+        )

+    def test_valid_signature_and_binding_are_not_execution_permission(self):

+        self.assertFalse(decision["execution_enabled"])
+        self.assertFalse(decision["mcp_claim_verified"])

+    def test_wrong_issuer_and_audience_refused(self):

+# V0.2 — Vérification cryptographique d'attestation externe (PR #2)
+
+**État : code vérificateur réel mais contrat hors ligne, non raccordé à un fournisseur d'identité ni au MCP runtime. Aucun service Web, scanner, accès source P2, correction ou déploiement activé.**
+
+Le module `app/signed_attestation.py` valide une attestation **émise à l'extérieur** au format JWS Ed25519 (`alg=EdDSA`), sans créer de signataire, autorité, jeton ni droit. La clé publique de confiance doit provenir de la configuration serveur protégée, **jamais du corps de la requête, d'un JWT `jwk/jku`, ni d'un utilisateur**. L'émetteur et l'audience acceptés doivent également être épinglés côté serveur.
+
+La validation vérifie l'algorithme et le `kid` explicitement autorisés, la signature, les champs stricts et non dupliqués, `iss`, `aud`, `sub`, les identifiants du projet et de l'actif, la portée P1/P2, l'opération exacte, `iat`/`nbf`/`exp` (durée ≤300 secondes) et le statut de révocation fourni par un mécanisme **externe fiable**. Les mutations des claims, expirations, actions inconnues, extra `admin`, signatures malformées, jetons trop longs et checker de révocation absent/indisponible sont rejetés sans exposer les données signées.
+
+**Séparation des portes :**
+
+1. Authentification de session (IdP réel ; actuellement ABSENTE).
+2. Vérification cryptographique d'une attestation externe (ce prototype hors ligne, disponible).
+3. Admission de projet/ressource et délégation P1/P2, y compris révocation fiable (NON intégrées).
+4. Claim MCP runtime, quotas, protections réseau, validation PHPPATCH et approbations (NON intégrés).
+5. Exécution de l'opération (toujours interdite).
+
+Le résultat du vérificateur, y compris pour une signature correcte, contient `authorized=false`, `execution_enabled=false`, `mcp_claim_verified=false`, `user_session_authenticated=false` et `project_acl_verified=false`. Il ne remplace pas `evaluate_action_gates` et ne doit **jamais** être transformé en décision `ALLOW` par un frontend ou un worker.
+
+**CI indépendante :** `.github/workflows/auth-contract.yml`, installation isolée de `cryptography==46.0.3` depuis `requirements-auth-contract.txt`, puis `auth_tests`. La CI historique sans dépendance et la CI FastAPI existante restent inchangées. La clé privée dans les tests est éphémère, créée uniquement pour valider les cas nominaux et négatifs ; aucune clé privée n'est ajoutée au dépôt.
+
+**Limite de sécurité majeure :** l'existence du vérificateur ne prouve pas que le MCP actuellement déployé émette de telles attestations ou dispose d'un mécanisme d'autorisation signé compatible. Il faut le vérifier et l'admettre auprès des autorités existantes avant tout raccordement de production. Aucune nouvelle autorité n'est créée.
