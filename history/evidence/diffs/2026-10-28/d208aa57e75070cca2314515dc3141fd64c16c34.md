+        with patch.dict(os.environ, {"P2S_ENABLE_LOCAL_GUEST_SUMMARY": "0"}):

+                    "url": "https://example.com/"

+    def test_public_peer_is_not_accepted_even_with_flag(self):
+        with patch.dict(os.environ, {"P2S_ENABLE_LOCAL_GUEST_SUMMARY": "1"}):
+            with client("192.0.2.55") as api:
+                r = api.post("/api/v0.2/guest/summary", json={
+                    "url": "https://example.com/"
+                })
+        self.assertEqual(r.status_code, 403)
+
+    def test_offline_anonymous_url_only_returns_honest_short_status(self):
+        with patch.dict(os.environ, {"P2S_ENABLE_LOCAL_GUEST_SUMMARY": "1"}):
+            with patch.object(socket.socket, "connect",
+                              side_effect=AssertionError("unexpected network")):
+                with client() as api:

+                        "url": "https://example.com/"
