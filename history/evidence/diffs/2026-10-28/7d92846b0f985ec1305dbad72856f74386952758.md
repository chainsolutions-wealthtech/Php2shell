+"""Offline PHPPATCH review packet: bind audit, handoff, diff and rehearsal.
+
+Only local user-supplied project files are read. The packet contains hashes
+and fixed allowlisted metadata; it cannot approve a patch, expose source,
+run PHP, create a PR, or modify any project/production file.
+"""
+from hashlib import sha256
+import json
+import re
+
+from .audit_dossier import build_audit_dossier
+from .dossier_handoff import prepare_phppatch_handoff
+from .patch_preview import propose_local_ini_patch
+from .patch_rehearsal import rehearse_local_ini_patch
+
+_ID = re.compile(r"P2S-EV-[0-9a-f]{16}\Z", re.ASCII)
+
+
+def build_patch_review_packet(project_path, as_of_utc, evidence_id):
+    """Create a deterministic, non-authoritative PHPPATCH review-only packet."""
+    if not isinstance(evidence_id, str) or not _ID.fullmatch(evidence_id):
+        raise ValueError("Invalid PHPPATCH evidence identifier")
+    dossier = build_audit_dossier(project_path, as_of_utc)
+    handoff = prepare_phppatch_handoff(dossier)
+    candidate_rows = [
+        item for item in handoff["review_items"]
+        if item["evidence_id"] == evidence_id
+    ]
+    if len(candidate_rows) != 1:
+        raise ValueError("Evidence is not present in the current handoff")
+    if candidate_rows[0]["indicator_rule"] != "PHP-CONFIG-DISPLAY-ERRORS":
+        raise ValueError("Only unambiguous local php.ini display_errors is supported")
+    proposal = propose_local_ini_patch(project_path, as_of_utc, evidence_id)
+    rehearsal = rehearse_local_ini_patch(project_path, as_of_utc, evidence_id)
+    fingerprint = dossier["report_fingerprint_sha256"]
+    if (handoff.get("source_dossier_fingerprint_sha256") != fingerprint
+            or proposal.get("source_dossier_fingerprint_sha256") != fingerprint
+            or rehearsal.get("source_dossier_fingerprint_sha256") != fingerprint):
+        raise ValueError("Dossier or diff lineage mismatch")
+    if (proposal["diff_sha256"] != rehearsal["proposed_diff_sha256"]
+            or rehearsal["source_evidence_id"] != evidence_id
+            or proposal["source_evidence_id"] != evidence_id):
+        raise ValueError("Proposal or rehearsal mismatch")
+    if (proposal["relative_file"] != rehearsal["relative_file"]
+            or proposal["patch_applied"] is not False
+            or rehearsal["patch_applied"] is not False
+            or rehearsal["source_files_written"] != 0
+            or rehearsal["in_memory_rollback_matches_original"] is not True
+            or rehearsal["exactly_one_line_changed"] is not True):
+        raise ValueError("Unsafe candidate or unverified in-memory rollback")
+    if (handoff["ready_for_production"] is not False
+            or proposal["ready_for_production"] is not False
+            or rehearsal["ready_for_production"] is not False):
+        raise ValueError("Elevated production capability")
+    # Only fixed fields are permitted in the packet: no raw diffs, configuration
+    # contents, credentials, absolute repository path or user mandate tokens.
+    proof = {
+        "source_dossier_sha256": fingerprint,
+        "evidence_id": evidence_id,
+        "relative_file": proposal["relative_file"],
+        "proposal_diff_sha256": proposal["diff_sha256"],
+        "source_file_sha256": rehearsal["source_file_sha256"],
+        "candidate_file_sha256": rehearsal["candidate_file_sha256"],
+        "in_memory_rollback_sha256": rehearsal["in_memory_rollback_sha256"],
+    }
+    packet_digest = sha256(json.dumps(
+        proof, sort_keys=True, separators=(",", ":")
+    ).encode("utf-8")).hexdigest()
+    return {
+        "schema_version": "0.3-preview",
+        "packet_type": "PHPPATCH_OFFLINE_REVIEW_PACKET_NOT_APPROVAL",
+        "as_of_utc": as_of_utc,
+        "proof": proof,
+        "packet_sha256": packet_digest,
+        "packet_sha256_is_signature": False,
+        "evidence_reconciled_against_current_local_dossier": True,
+        "single_ini_change_simulated_in_memory": True,
+        "rollback_in_memory_exact": True,
+        "source_files_written": 0,
+        "patch_applied": False,
+        "php_executed": False,
+        "sandbox_tests_executed": False,
+        "cve_applicability_confirmed": False,
+        "identity_and_mandate_verified": False,
+        "source_repository_access_authorized": False,
+        "independent_approval_received": False,
+        "git_pull_request_created": False,
+        "production_rollback_proven": False,
+        "ready_for_production": False,
+        "next_gates": [
+            "Verify actual project ownership, mandate and P2 asset/source scope.",
+            "Authenticate original audit evidence and effective HTTP PHP SAPI.",

+            "Review actual before/after behavior, maintain approved rollback.",
+            "Obtain independent review and release via existing governed GitHub/MCP.",
