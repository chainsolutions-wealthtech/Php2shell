+"""Strict offline revalidation of a PHPPATCH review-only packet.
+
+Recompute the packet from current local evidence; never treat a matching
+digest, packet, or user-provided JSON as identity, mandate or approval.
+No network, PHP execution, patch application or production mutation.
+"""
+import json
+from pathlib import Path
+
+from .patch_review_packet import build_patch_review_packet
+
+_MAX_PACKET_BYTES = 16_384
+
+
+def _canonical(value):
+    try:
+        serialized = json.dumps(
+            value, sort_keys=True, separators=(",", ":"), ensure_ascii=True,
+            allow_nan=False,
+        )
+    except (TypeError, ValueError, OverflowError) as exc:
+        raise ValueError("Invalid local review packet") from exc
+    if len(serialized.encode("utf-8")) > _MAX_PACKET_BYTES:
+        raise ValueError("Local review packet exceeds size limit")
+    return serialized
+
+
+def _reject_duplicate_keys(pairs):
+    obj = {}
+    for key, value in pairs:
+        if key in obj:
+            raise ValueError("Duplicate JSON field in local review packet")
+        obj[key] = value
+    return obj
+
+
+def _reject_json_constant(_value):
+    raise ValueError("Non-finite JSON value in local review packet")
+
+
+def verify_local_review_packet(packet, project_path, as_of_utc, evidence_id):
+    """Compare *every* field against a fresh local, non-authoritative packet.
+
+    A hash alone is not sufficient: this comparison also rejects added fields,
+    elevated capability flags, stale evidence and JSON type substitutions.
+    """
+    if not isinstance(packet, dict):
+        raise ValueError("Local review packet must be a JSON object")
+    supplied = _canonical(packet)
+    expected = build_patch_review_packet(project_path, as_of_utc, evidence_id)
+    if supplied != _canonical(expected):
+        raise ValueError("Local review packet does not match current evidence")
+    return {
+        "schema_version": "0.3-preview",
+        "verification_scope": "LOCAL_REVIEW_PACKET_MATCH_ONLY_NOT_AUTHORITY",
+        "packet_sha256": expected["packet_sha256"],
+        "local_lineage_matches": True,
+        "packet_signature_verified": False,
+        "identity_and_mandate_verified": False,
+        "independent_approval_received": False,
+        "patch_applied": False,
+        "sandbox_tests_executed": False,
+        "production_rollback_proven": False,
+        "ready_for_production": False,
+    }
+
+
+def load_and_verify_local_review_packet(packet_file, project_path, as_of_utc,
+                                        evidence_id):
+    """Load a bounded, regular local JSON file without following symlinks."""
+    if not isinstance(packet_file, (str, Path)):
+        raise ValueError("Invalid local review packet file")
+    path = Path(packet_file)
+    if any(part.is_symlink() for part in (path, *path.parents)):
+        raise ValueError("Symlinked local review packet path is forbidden")
+    if not path.is_file() or path.stat().st_size > _MAX_PACKET_BYTES:
+        raise ValueError("Local review packet unavailable or oversized")
+    data = path.read_bytes()
+    if len(data) > _MAX_PACKET_BYTES:
+        raise ValueError("Local review packet exceeds size limit")
+    try:
+        packet = json.loads(
+            data.decode("utf-8"),
+            object_pairs_hook=_reject_duplicate_keys,
+            parse_constant=_reject_json_constant,
+        )
+    except (UnicodeError, json.JSONDecodeError) as exc:
+        raise ValueError("Invalid local review packet JSON") from exc
+    return verify_local_review_packet(
+        packet, project_path, as_of_utc, evidence_id,
+    )

+# V0.3 — Revalidation locale d'un paquet PHPPATCH (non autorisante)
+
+Le module `app.patch_packet_verifier` complète le paquet de revue
+`app.patch_review_packet` sans exécuter de correctif. Il recalcule le dossier
+d'audit local et **compare l'intégralité du paquet** fourni avec la version
+reconstruite depuis les fichiers actuels, pas uniquement son empreinte.
+
+## Utilisation hors ligne
+
+```python
+from app.patch_packet_verifier import load_and_verify_local_review_packet
+
+result = load_and_verify_local_review_packet(
+    "/chemin/local/packet.json",
+    "/chemin/local/projet-autorise",
+    "2026-10-28",
+    "P2S-EV-0123456789abcdef",
+)
+```
+
+L'identifiant doit provenir du dossier d'audit courant ; l'exemple est fictif.
+La fonction accepte également un objet Python via
+`verify_local_review_packet(packet, project_path, as_of_utc, evidence_id)`.
+
+## Garde-fous
+
+- Lecture de fichier JSON local uniquement, limite 16 Kio, rejet des liens
+  symboliques, des clés JSON dupliquées et des constantes non finies.
+- Comparaison canonique des **types et champs complets** : toute modification,
+  promotion de drapeau, champ supplémentaire ou dossier obsolète est refusé.
+- Les erreurs ne reproduisent ni configuration PHP, ni source, ni secrets.
+- Les tests vérifient le refus des altérations, l'absence de réseau et
+  l'absence d'écriture sur les fichiers du projet.
+
+## Frontières de confiance
+
+Un paquet correspondant aux fichiers locaux **n'est pas signé** et ne prouve
+ni identité, ni mandat P1/P2, ni propriété de domaine, ni approbation, ni
+compatibilité PHP/SAPI, ni test Docker, ni correction d'une CVE. La sortie
+conserve `ready_for_production=false`, `patch_applied=false` et
+`identity_and_mandate_verified=false`. Le contrôle ne contacte aucun
+site, ne crée aucune PR, ne modifie ni `main` ni S1.
+
+Cette tranche constitue une preuve **partielle** pour le transfert et la
+revue PHPPATCH (REQ-030, P2S-WF-R06) et la non-régression (P2S-WF-Q01).
+Elle ne remplace pas les gates réels de la roadmap V0.1→V1.0.

+from copy import deepcopy
+import json
+import socket
+import tempfile
+import unittest
+from pathlib import Path
+from unittest.mock import patch
+
+from app.audit_dossier import build_audit_dossier
+from app.patch_review_packet import build_patch_review_packet
+from app.patch_packet_verifier import (
+    verify_local_review_packet, load_and_verify_local_review_packet,
+)

+
+class PatchPacketVerifierTests(unittest.TestCase):
+    def setUp(self):
+        self.temp = tempfile.TemporaryDirectory()
+        self.addCleanup(self.temp.cleanup)
+        self.root = Path(self.temp.name)
+        self.target = self.root / "php.ini"
+        self.target.write_bytes(
+            b"display_errors=On\n"
+            b"session.secret=PRIVATE_NEVER_EXPORT\n"
+        )
+        dossier = build_audit_dossier(self.root, "2026-10-28")
+        self.evidence = next(
+            row["evidence_id"] for row in dossier["observations"]
+            if row["rule"] == "PHP-CONFIG-DISPLAY-ERRORS"
+        )
+        self.packet = build_patch_review_packet(
+            self.root, "2026-10-28", self.evidence,
+        )
+
+    def verify(self, packet):
+        return verify_local_review_packet(
+            packet, self.root, "2026-10-28", self.evidence,
+        )
+
+    def test_current_packet_matches_without_authorizing_anything(self):
+        before = self.target.read_bytes()
+        with patch.object(socket.socket, "connect", side_effect=AssertionError("network")):
+            result = self.verify(self.packet)
+        self.assertEqual(self.target.read_bytes(), before)
+        self.assertTrue(result["local_lineage_matches"])
+        self.assertEqual(result["packet_sha256"], self.packet["packet_sha256"])
+        for field in (
+            "packet_signature_verified", "identity_and_mandate_verified",
+            "independent_approval_received", "patch_applied",
+            "sandbox_tests_executed", "production_rollback_proven",
+            "ready_for_production",
+        ):
+            with self.subTest(field=field):
+                self.assertIs(result[field], False)
+        self.assertNotIn("PRIVATE_NEVER_EXPORT", json.dumps(result))
+        self.assertNotIn(str(self.root), json.dumps(result))
+
+    def test_tampered_proof_and_elevated_flags_are_rejected(self):
+        for field, value in (
+            ("ready_for_production", True),
+            ("ready_for_production", 0),
+            ("identity_and_mandate_verified", True),
+        ):
+            with self.subTest(field=field, value=value):
+                forged = deepcopy(self.packet)
+                forged[field] = value
+                with self.assertRaises(ValueError):
+                    self.verify(forged)
+        forged = deepcopy(self.packet)
+        forged["proof"]["source_file_sha256"] = "0" * 64
+        with self.assertRaises(ValueError):
+            self.verify(forged)
+
+    def test_extra_fields_and_type_substitutions_are_rejected(self):
+        forged = deepcopy(self.packet)
+        forged["operator_authorized"] = True
+        with self.assertRaises(ValueError):
+            self.verify(forged)
+        forged = deepcopy(self.packet)
+        forged["source_files_written"] = 0.0
+        with self.assertRaises(ValueError):
+            self.verify(forged)
+        with self.assertRaises(ValueError):
+            self.verify([self.packet])

+    def test_stale_source_cannot_reuse_prior_packet(self):

+            b"display_errors=On\n"
